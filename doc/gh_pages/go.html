Embedding cameo in a Go program

The package github.com/cameo-lang/cameo/src/cameo loads a script, runs it, calls its methods and converts values both ways.

Adding cameo to a module

cameo needs Go 1.26.4 or later; go get raises a module's go line to 1.26.4. Import the package, then:

go get github.com/cameo-lang/cameo@v0.2.1
go mod tidy

go mod tidy adds cameo's own dependencies, google.golang.org/protobuf and github.com/go-ruby-parser/parser, to go.sum; without it the build fails with "missing go.sum entry". Neither needs cgo.

Load, Run and RunContext

cameo.Load(file, src, stdout, stderr) parses and checks a script and returns a *cameo.Program; file names it in error messages, and the script's puts and warn write to stdout and stderr. Run evaluates the script once, and its classes and methods stay loaded for Call. LastValue returns the script's last top-level expression as p shows it. cameo.Run does Load and Run in one call, as the cameo command does.

package main

import (
	"fmt"
	"log"
	"os"

	"github.com/cameo-lang/cameo/src/cameo"
)

const script = `
class Point
  int :x, :y

  def +(other) = Point.new(x: x + other.x, y: y + other.y)
end

def greet(name) = "hello, #{name}"

puts "loaded"
Point.new(x: 1, y: 2)
`

func main() {
	prog, err := cameo.Load("point.cameo", script, os.Stdout, os.Stderr)
	if err != nil {
		log.Fatal(err)
	}
	if err := prog.Run(); err != nil {
		log.Fatal(err)
	}
	last, _ := prog.LastValue()
	fmt.Println("last value:", last)

	hello, err := prog.Call(nil, "greet", "Go")
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(hello)

	pt, err := prog.Call("Point", "new", map[string]any{"x": 1, "y": 2})
	if err != nil {
		log.Fatal(err)
	}
	sum, err := prog.Call(pt, "+", pt)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Printf("%T\n", sum)
	x, _ := prog.Call(sum, "x")
	fmt.Printf("%T %v\n", x, x)
}

RunContext(ctx) is Run until the context ends. Then the script, and every worker it started, stops at its next loop pass, method call or block call, no rescue catches it, and RunContext returns the context's error.

package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/cameo-lang/cameo/src/cameo"
)

func main() {
	prog, err := cameo.Load("spin.cameo", "loop { }", os.Stdout, os.Stderr)
	if err != nil {
		log.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
	defer cancel()
	err = prog.RunContext(ctx)
	fmt.Println(err)
}
context deadline exceeded

A host method that blocks is not interrupted: the script stops once it returns, so a method that waits on something should watch the same context.

Calling into a script

prog.Call(recv, method, args...) calls a method and returns its result as a Go value. It runs the script first if Run has not. The receiver is one of:

  • nil, for the script's top-level methods: prog.Call(nil, "greet", "Go");
  • a class name, for a class method: prog.Call("Geo::Point", "new", …);
  • a proto.Message of a loaded class, such as an instance a previous Call returned;
  • any other Go value ToValue converts: prog.Call("abc", "upcase") returns "ABC". A string that names a loaded class is that class.

The arguments are converted with ToValue, and the result with ToGo. Call passes positional arguments only; a class without initialize also takes its fields as a map[string]any, as the example above does.

Any other Go value is an error: "cannot convert struct {} to a cameo value". A Hash's Symbol keys become strings in Go, as a Symbol does wherever a value leaves cameo. A Hash that has no Go map form is an error: one with both :a and "a" as keys, or with keys that are not comparable in Go, such as a Range or a binary String.

Protobuf messages both ways

Every class a script defines is a protobuf message, with one field per declared field, numbered from a hash of its name, or by the field's number:. A class inside module Geo is the message Point in package geo.

  • prog.Proto(name) returns the .proto text of one class.
  • prog.ProtoFiles() returns the .proto files of every loaded class, one per package and keyed by file name, with the value envelope in cameo/value.proto. Compile them with protoc to get Go types with the same wire form.
  • prog.Descriptor(name) returns the class's protoreflect.MessageDescriptor, from which dynamicpb builds a message without generated code.
  • prog.Encode(v) writes any value, not only an instance, in the cameo.Value envelope; prog.Decode reads it back.
  • protogen.Merge(registered, next) merges two definitions of one message, as two programs that share a store have: the fields both have stay as registered, and the fields only next has are added, numbered from their names. The design document says what it accepts.
package main

import (
	"fmt"
	"io"
	"log"
	"maps"
	"slices"

	"google.golang.org/protobuf/proto"
	"google.golang.org/protobuf/reflect/protoreflect"
	"google.golang.org/protobuf/types/dynamicpb"

	"github.com/cameo-lang/cameo/src/cameo"
)

const script = `
module Geo
  class Point
    int :x, :y

    def norm2 = x * x + y * y
  end
end
`

func check(err error) {
	if err != nil {
		log.Fatal(err)
	}
}

func main() {
	prog, err := cameo.Load("geo.cameo", script, io.Discard, io.Discard)
	check(err)
	check(prog.Run())

	// The .proto text of one class, and the files of all of them.
	text, err := prog.Proto("Geo::Point")
	check(err)
	fmt.Print(text)
	files := prog.ProtoFiles()
	fmt.Println(slices.Sorted(maps.Keys(files)))

	// A message built in Go from the class's descriptor is an instance in
	// the script.
	desc, err := prog.Descriptor("Geo::Point")
	check(err)
	msg := dynamicpb.NewMessage(desc)
	msg.Set(desc.Fields().ByName("x"), protoreflect.ValueOfInt64(3))
	msg.Set(desc.Fields().ByName("y"), protoreflect.ValueOfInt64(4))
	n, err := prog.Call(msg, "norm2")
	check(err)
	fmt.Println(n)

	// An instance comes back as a proto.Message.
	pt, err := prog.Call("Geo::Point", "new", map[string]any{"x": 1, "y": 2})
	check(err)
	wire, err := proto.Marshal(pt.(proto.Message))
	check(err)
	fmt.Println(len(wire), "bytes")

	// Any value travels in the cameo.Value envelope.
	data, err := prog.Encode([]any{"a", int64(1), pt})
	check(err)
	back, err := prog.Decode(data)
	check(err)
	fmt.Println(len(back.([]any)), "elements")
}

Errors

A script that does not check returns a syntax.ErrorList (package github.com/cameo-lang/cameo/src/syntax) from Load, with every error and its line. An exception that ends Run or Call returns a *cameo.RunError, whose Message is formatted as the cameo command prints it and whose Unwrap returns the exception, an *object.Exception.

package main

import (
	"errors"
	"fmt"
	"io"

	"github.com/cameo-lang/cameo/src/cameo"
	"github.com/cameo-lang/cameo/src/syntax"
)

func main() {
	_, err := cameo.Load("bad.cameo", "@count = 1\n$total = 2\n", io.Discard, io.Discard)
	var list syntax.ErrorList
	if errors.As(err, &list) {
		for _, e := range list {
			fmt.Printf("line %d: %s\n", e.Line, e.Msg)
		}
	}

	err = cameo.Run("boom.cameo", "def half(n) = n / 0\nhalf(4)\n", io.Discard, io.Discard)
	var runErr *cameo.RunError
	if errors.As(err, &runErr) {
		fmt.Println(runErr.Message)
	}
}

line 2: global variables are not supported

Concurrency

Inside a script that includes Cameo, async runs a method call on a goroutine of its own, over copies of its receiver and arguments; see async, await and promises. Every worker reaches the host's modules, so a host method may run on several goroutines at once and must be safe for that. RunContext stops the workers with the script.

package main

import (
	"fmt"
	"log"
	"os"
	"sync/atomic"

	"github.com/cameo-lang/cameo/src/cameo"
)

const script = `
include Cameo

class Fetcher
  string :base

  def get(path) = Http.get(base + path)
end

f = Fetcher.new(base: "https://example.com")
jobs = ["/a", "/b", "/c"].map { |path| async f, :get, path }
p jobs.map { |job| await job }
`

func main() {
	prog, err := cameo.Load("fetch.cameo", script, os.Stdout, os.Stderr)
	if err != nil {
		log.Fatal(err)
	}
	var calls atomic.Int64
	prog.Module("Http").Def("get", func(args []any, kwargs map[string]any) (any, error) {
		calls.Add(1) // runs on each worker's goroutine
		return "body of " + args[0].(string), nil
	})
	if err := prog.Run(); err != nil {
		log.Fatal(err)
	}
	fmt.Println(calls.Load(), "calls")
}
["body of https://example.com/a", "body of https://example.com/b", "body of https://example.com/c"]
3 calls

Promises: I/O in the host

cameo does no I/O. A host method that does some, such as fetching a web page, can return a promise instead of blocking: it makes one with prog.Promise(), starts the work on a goroutine, and returns the promise at once. The script holds the promise as an Integer and awaits it; the goroutine settles it with Resolve, with a Go value that converts as a host method's result does, or with Fail, with an error that await raises. A script that starts every fetch before it awaits any has the requests run at the same time. Resolve and Fail may be called from any goroutine, once; a second call returns cameo.ErrSettled. This is examples/webfetch:

// Command webfetch fetches web pages from a cameo script. cameo does no
// I/O, so the host gives the script a Web module whose fetch starts the
// request on a goroutine and returns a promise at once. The script starts
// every fetch before it awaits any, so the requests run at the same time,
// and await returns each page or raises Web::FetchError.
//
//	go run ./examples/webfetch https://example.com https://example.org/missing
package main

import (
	"fmt"
	"io"
	"net/http"
	"os"

	"github.com/cameo-lang/cameo/src/cameo"
)

const script = `
include Cameo

def fetch_all(urls)
  promises = urls.map { |url| Web.fetch(url) }   # every request starts now
  urls.zip(promises).map do |url, promise|
    "#{url}: #{(await promise).size} bytes"
  rescue Web::FetchError => e
    "#{url}: #{e.message}"
  end
end
`

// fetchAll runs the script's fetch_all over urls with client.
func fetchAll(client *http.Client, urls []string) ([]any, error) {
	prog, err := cameo.Load("webfetch.cameo", script, os.Stdout, os.Stderr)
	if err != nil {
		return nil, err
	}
	web := prog.Module("Web")
	fetchError := web.Exception("FetchError")
	web.Def("fetch", func(args []any, kwargs map[string]any) (any, error) {
		url, ok := args[0].(string)
		if !ok {
			return nil, fmt.Errorf("Web.fetch takes a URL, not %v", args[0])
		}
		promise := prog.Promise()
		go func() {
			resp, err := client.Get(url)
			if err != nil {
				promise.Fail(fetchError.New("%v", err))
				return
			}
			defer resp.Body.Close()
			if resp.StatusCode != http.StatusOK {
				promise.Fail(fetchError.New("%s", resp.Status))
				return
			}
			body, err := io.ReadAll(resp.Body)
			if err != nil {
				promise.Fail(fetchError.New("%v", err))
				return
			}
			promise.Resolve(string(body))
		}()
		return promise, nil
	})
	lines, err := prog.Call(nil, "fetch_all", urls)
	if err != nil {
		return nil, err
	}
	return lines.([]any), nil
}

func main() {
	lines, err := fetchAll(http.DefaultClient, os.Args[1:])
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	for _, line := range lines {
		fmt.Println(line)
	}
}
